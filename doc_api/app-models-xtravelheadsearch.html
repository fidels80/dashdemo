                public static function findWithMaxRowDate()
{
    return self::find()
        ->alias('h')
        ->leftJoin('xtravelrow r', 'r.th_id = h.th_id')
        ->addSelect([
            'h.*',
            'max_data' => new \yii\db\Expression("
                MAX(
                    CASE 
                        WHEN r.check_out > r.check_in THEN r.check_out
                        ELSE r.check_in
                    END
                )
            ")
        ])
        ->groupBy('h.th_id');
}
