                public function Analisiparty($id)
{
    $query = new Query;
    $query->select([
        'party',
        //' case when citta is null then citta_da else citta end as citta    ',
        new \yii\db\Expression("sum(tax_unit*qta) as ttassa"),
        new \yii\db\Expression("sum(imponibile) as timponibile"),
        new \yii\db\Expression("sum(iva) as tiva"),
        new \yii\db\Expression("sum(fee) as tfee"),
        new \yii\db\Expression("sum(totale) as ttotale"),
    ])
        ->from('adb_auxcoop.dbo.xtravelrow')
        ->leftJoin('adb_auxcoop.dbo.AR', 'AR.Cd_AR = xtravelrow.cd_Ar')
        ->where([
            'and',
            ['th_id' => $id],
            ['AR.x_isacconto' => null]
        ])
        ->groupBy([
            'party',
            //'case when citta is null then citta_da else citta end as citta',
        ])
        ->orderBy(['party' => SORT_ASC]); // Aggiungiamo l'ordinamento qui
    $command = $query->createCommand();
    $data           = $command->queryAll();
    $out = array_values($data);
    $labels = [];
    $series = [];
    foreach ($out as $value) {
        $labels[] = $value['party'];
        $series[] = $value['ttassa'] + $value['ttotale'];
    };
    return $out;
}

                public function actionCrea()
{
    $model = new Xtravelhead();
    $db = Xtravelhead::getDb();
    if ($model->load(Yii::$app->request->post())) {
        // Pulizia campi calcolati e default
        unset($model->decodificatore);
        $model->bloccato = 0;
        $model->fatturato = 0;
        $model->evaso_A = 0; // Inizializziamo a 0 per evitare errori di validazione
        $model->evaso_p = 0;
        // Passiamo le date nel formato gradito al tuo SQL Server (Giorno-Mese-Anno Ora:Minuti:Secondi)
        $model->datath = date('d-m-Y H:i:s');
        $model->x_maxdata = date('d-m-Y H:i:s', strtotime('+30 years'));
        // 1. Salvataggio Testata
        if ($model->save()) {
            // 2. Salvataggio Tour Managers (Tabella di giunzione)
            $dbManagers = Yii::$app->db5;
            $tableName = '[adb_auxcoop].[dbo].[xtravel_managers]';
            try {
                if (is_array($model->manager_ids) && !empty($model->manager_ids)) {
                    $rows = [];
                    foreach ($model->manager_ids as $userId) {
                        if (!empty($userId)) {
                            $rows[] = [(int)$model->th_id, (int)$userId];
                        }
                    }
                    $dbManagers->createCommand()->batchInsert($tableName, ['th_id', 'user_id'], $rows)->execute();
                }
                Yii::$app->session->setFlash('success', "Prenotazione n. {$model->numero} creata con successo!");
                return $this->redirect(['index']);
            } catch (\Exception $e) {
                // Se fallisce il manager, avvisiamo ma la testata è salva
                Yii::$app->session->setFlash('warning', "Prenotazione creata, ma errore nel salvataggio dei manager: " . $e->getMessage());
                return $this->redirect(['index']);
            }
        } else {
            // Flash errore validazione
            $errori = implode("<br>", \yii\helpers\ArrayHelper::flatten($model->getErrors()));
            Yii::$app->session->setFlash('error', "Errore nella creazione: <br>" . $errori);
        }
    } else {
        // Logica nuovo numero
        $nuovoNumero = (new \yii\db\Query())
            ->select(['n' => 'ISNULL(MAX(CAST(numero AS INT)), 0) + 1'])
            ->from('xtravelhead')
            ->scalar($db);
        $model->numero = (string)$nuovoNumero;
    }
    // Clienti per la form
    $clienti = (new \yii\db\Query())
        ->select(['Cd_Cf', 'Descrizione'])
        ->from('Cf')
        ->all($db);
    return $this->render('crea', [
        'model' => $model,
        'clienti' => $clienti,
    ]);
}
